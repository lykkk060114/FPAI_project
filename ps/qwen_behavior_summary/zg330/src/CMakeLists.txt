cmake_minimum_required(VERSION 3.16)
project(qwen_icraft LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS ON)
set(ICRAFT_SDK /tmp/sdk/usr CACHE PATH "Extracted iCraft SDK")
add_executable(qwen_icraft qwen_icraft.cpp)
target_compile_options(qwen_icraft PRIVATE -O2)
target_include_directories(qwen_icraft PRIVATE ${ICRAFT_SDK}/include)
target_link_directories(qwen_icraft PRIVATE ${ICRAFT_SDK}/lib/aarch64-linux-gnu)
target_link_options(qwen_icraft PRIVATE -Wl,-rpath-link,${ICRAFT_SDK}/lib/aarch64-linux-gnu)
target_link_libraries(qwen_icraft PRIVATE icraft_xrt icraft_xir icraft_hostbackend icraft_zg330backend icraft_utils pthread dl)
